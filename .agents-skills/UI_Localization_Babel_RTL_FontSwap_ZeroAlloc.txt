## SECTION 1 — EXECUTION POLICY

### [RULE-LOC-001] Zero-Alloc Lookup
All localization keys MUST resolve via `static readonly int` hashes. String-based dictionary
access is PROHIBITED in any hot-path (Update, HUD render, damage number display).

// COMPLIANT
public static class LocKeys
{
    public static readonly int HUD_HULL_LABEL       = LocHash.Compute("hud.hull.label");
    public static readonly int HUD_SHIELD_LABEL     = LocHash.Compute("hud.shield.label");
    public static readonly int NARRATIVE_INTRO_00   = LocHash.Compute("narrative.intro.00");
    public static readonly int WORLD_SECTOR_KORRATH = LocHash.Compute("world.sector.korrath");
}

// Lookup — zero alloc, returns ref into pool buffer
ReadOnlySpan<char> label = LocRegistry.Resolve(LocKeys.HUD_HULL_LABEL);

// NON-COMPLIANT — string key at runtime
var label = LocRegistry.Resolve("hud.hull.label"); // [FORBID]

### [RULE-LOC-002] Staged Asset Swap — Max 18 Labels Per Tick
When a language change triggers font-atlas swap, UI label re-assignment MUST be batched.
Maximum throughput: 18 TMP_Text registrations per engine Tick.
Excess registrations are enqueued and drained across subsequent Ticks.

public sealed class LabelSwapScheduler
{
    private const int MAX_PER_TICK = 18;
    private readonly Queue<TMP_TextEntry> _pending = new Queue<TMP_TextEntry>(256);

    // Called once per Tick by LocalizationCoordinator
    public void DrainTick(Material newMaterial, TMP_FontAsset newFont)
    {
        int processed = 0;
        while (_pending.Count > 0 && processed < MAX_PER_TICK)
        {
            TMP_TextEntry entry = _pending.Dequeue();
            entry.Text.font         = newFont;
            entry.Text.fontMaterial = newMaterial;
            entry.Text.SetText(LocRegistry.ResolveRaw(entry.KeyHash));
            processed++;
        }
    }

    public void Enqueue(TMP_TextEntry entry) => _pending.Enqueue(entry);
}

### [FORBID-LOC-001] No `string.Format` in HUD
`string.Format`, `String.Concat`, interpolated strings (`$"..."`) are FORBIDDEN in HUD
render paths. Use pre-allocated numeric placeholder templates backed by `LocNumericBuffer`.

// NON-COMPLIANT
hudText.text = $"HULL: {hullValue}/{maxHull}";          // [FORBID] — allocates
hudText.text = string.Format("HULL: {0}/{1}", h, m);   // [FORBID] — allocates

// COMPLIANT — template with pre-allocated buffer
// Template stored as: "HULL: {N0}/{N1}"
// Buffer reused per-frame
LocNumericBuffer.Write(LocKeys.HUD_HULL_LABEL, hullValue, maxHull, out ReadOnlySpan<char> result);
hudText.SetCharArray(result); // Zero-alloc TMP path

`LocNumericBuffer` Contract:
- Fixed `char[]` internal buffer, size = 128.
- Supports up to 4 numeric placeholders (`{N0}`–`{N3}`).
- Thread-safe via `[ThreadStatic]` buffer instance.

### [FORBID-LOC-002] No Runtime Font-Sweep via `FindObjectsOfType<TMP_Text>`
All `TMP_Text` instances MUST register to `TMP_TextRegistry` on `Awake`.
`FindObjectsOfType<TMP_Text>` is PERMANENTLY FORBIDDEN — causes full scene graph traversal,
produces O(n) GC pressure proportional to scene object count.

// NON-COMPLIANT
TMP_Text[] all = FindObjectsOfType<TMP_Text>(); // [FORBID]

// COMPLIANT — registry-driven enumeration
TMP_TextRegistry.ForEachInLayer(LocLayer.HUD, entry => {
    swapScheduler.Enqueue(entry);
});

## SECTION 2 — DATA ARCHITECTURE

### [DATA-LOC-001] Multi-Layer String Pool Architecture
String pools are partitioned into three isolation layers to control memory residency and
eviction granularity independently.

[STRUCT_START]
                    LOCALIZATION STRING POOL
├
  LAYER: CORE     LAYER: WORLD          LAYER: NARRATIVE

 HUD labels     Sector names           Cutscene dialogue
 System msgs    NPC faction titles     Codex entries
 Error strings  Item/weapon names      Mission briefings
 Menu text      Environmental hints    Terminal logs

 ALWAYS loaded  Loaded per zone        Streamed on demand
 ~48 KB max     ~256 KB max            ~2 MB max
 No eviction    Evict on zone exit     Evict on scene end
[STRUCT_END]

Implementation:

public enum LocLayer { Core, World, Narrative }

public static class LocRegistry
{
    // Each layer: Dictionary<int, NativeString128> for zero-alloc value storage
    private static readonly LocPool _core      = new LocPool(capacity: 512,   layer: LocLayer.Core);
    private static readonly LocPool _world     = new LocPool(capacity: 2048,  layer: LocLayer.World);
    private static readonly LocPool _narrative = new LocPool(capacity: 16384, layer: LocLayer.Narrative);

    public static ReadOnlySpan<char> Resolve(int keyHash)
    {
        if (_core.TryGet(keyHash, out var val))      return val.AsSpan();
        if (_world.TryGet(keyHash, out val))         return val.AsSpan();
        if (_narrative.TryGet(keyHash, out val))     return val.AsSpan();

        // [SAFE-LOC-001] Missing key fallback — see Section 5
        LocErrorLog.LogOnce(keyHash);
        return MissingKeyLiteral.AsSpan(); // "[ERR_MISSING_KEY]"
    }
}

### [DATA-LOC-002] Glitch Encoding — Binary-Noise Table for Text Decay
At high Hull Stress (threshold: ≥ 75% damage), HUD text undergoes Text Decay rendering.
A pre-baked binary-noise substitution table maps Unicode codepoints to glitch variants.
The table is static, immutable, and loaded at boot. Zero runtime allocation.

public static class GlitchEncoder
{
    // 256-entry lookup: char -> glitch char, pre-baked at build time
    // Values authored by art team in GlitchTable.bytes (StreamingAssets)
    private static readonly ushort[] _noiseTable = GlitchTableLoader.LoadStatic();

    // Stress: 0.0 = clean, 1.0 = full decay
    // decayProbability derived from: stress = clamp((hull_stress - 0.75) / 0.25, 0, 1)
    public static char ApplyDecay(char input, float decayProbability, uint frameSeed)
    {
        // Deterministic per-frame noise — no Random allocation
        uint hash = (uint)input ^ (frameSeed * 2246822519u);
        float noise = (hash & 0xFFFF) / 65535f;
        if (noise < decayProbability)
            return (char)_noiseTable[input & 0xFF];
        return input;
    }
}

Decay Activation Rule:
- Hull Stress < 75%  → No decay. Standard render.
- Hull Stress 75–90% → Partial decay. `decayProbability` = 0.0–0.5 (linear).
- Hull Stress > 90%  → Heavy decay. `decayProbability` = 0.5–1.0 (linear).
- Decay applied only to `LocLayer.Core` (HUD). Never applied to Narrative layer text.

## SECTION 3 — ALGORITHMIC & MATH TRUTH

### [MATH-LOC-001] RTL Reversal Algorithm — Zero-GC Arabic/Hebrew Reordering
RTL character reordering is performed via manual index-swap in a fixed char buffer.
`string.Reverse()`, LINQ reversal, and any method producing intermediate allocations are
FORBIDDEN for this operation.

public static class RTLProcessor
{
    // Fixed staging buffer — [ThreadStatic] for thread safety, no allocation
    [ThreadStatic]
    private static char[] _stagingBuffer;

    private static char[] GetBuffer()
        => _stagingBuffer ??= new char[1024]; // one-time alloc per thread

    /// <summary>
    /// Reverses a logical RTL string into visual order.
    /// Handles Arabic ligature boundaries and Hebrew vowel diacritics.
    /// Returns span into internal staging buffer — caller must consume before next call.
    /// </summary>
    public static ReadOnlySpan<char> ToVisualOrder(ReadOnlySpan<char> logical)
    {
        char[] buf = GetBuffer();
        int len    = logical.Length;

        // Phase 1: Copy to buffer
        logical.CopyTo(buf.AsSpan(0, len));

        // Phase 2: In-place reverse — O(n/2), zero alloc
        int left  = 0;
        int right = len - 1;
        while (left < right)
        {
            // Diacritic-safe swap: do not split Arabic combining marks
            if (IsArabicCombiningMark(buf[right]))
            {
                // Keep combining mark anchored to its base — shift right cursor left
                right--;
                continue;
            }
            if (IsArabicCombiningMark(buf[left]))
            {
                left++;
                continue;
            }
            (buf[left], buf[right]) = (buf[right], buf[left]);
            left++;
            right--;
        }

        // Phase 3: Return span — zero alloc, caller consumes immediately
        return buf.AsSpan(0, len);
    }

    // Unicode range check — no allocation
    private static bool IsArabicCombiningMark(char c)
        => c >= '\u0610' && c <= '\u061A'   // Arabic Extended
        || c >= '\u064B' && c <= '\u065F';  // Arabic Diacritics
}

Verification Requirement:
RTL reversal MUST pass the HECTON-8 RTL Conformance Test Suite (72 test strings covering
Arabic, Hebrew, bidirectional mixed segments) before merge to `main`.

### [FLOW-LOC-001] Language Reboot Sequence — 4-Stage Pipeline
Language change triggers a deterministic async pipeline. Stages are strictly ordered.
No stage may begin before its predecessor completes. Frame-spike prevention is enforced
via `LabelSwapScheduler` (see [RULE-LOC-002]).

LANGUAGE REBOOT PIPELINE

  STAGE 1 — LOAD FONT ADDRESSABLE
  [STRUCT_START]
   AsyncOperationHandle<TMP_FontAsset> handle =
       Addressables.LoadAssetAsync<TMP_FontAsset>(fontKey);

   await handle;  // Zero-GC Unity async awaiter.
   TMP_FontAsset newFont = handle.Result;

                               COMPLETE → proceed
                              ▼
  STAGE 2 — GENERATE MATERIAL INSTANCE
  [STRUCT_START]
   // Must run on main thread (Unity GPU API constraint)
   Material newMat = new Material(newFont.material);
   newMat.SetTexture("_MainTex", newFont.atlasTexture);
   newMat.EnableKeyword("UNITY_UI_CLIP_RECT");

                               COMPLETE → proceed
                              ▼
  STAGE 3 — BACKGROUND-ITERATE REGISTRY
  [STRUCT_START]
   // Enqueue all registered TMP_Text entries to SwapScheduler
   // SwapScheduler drains at MAX 18 per Tick (frame-safe)
   TMP_TextRegistry.ForEachAll(entry =>
       swapScheduler.Enqueue(entry));

   while (swapScheduler.HasPending)
       await NextTick(); // yield to engine loop

                               ALL LABELS SWAPPED → proceed
                              ▼
  STAGE 4 — RELEASE OLD ATLAS
  [STRUCT_START]
   Addressables.Release(previousFontHandle);
   // Deterministic ref-count release only. Never call Resources.UnloadUnusedAssets().
   previousFontHandle = handle; // track for next reboot
  [STRUCT_END]

  TOTAL EXPECTED DURATION:
    Stage 1: 80–400ms (network/disk dependent, async)
    Stage 2: < 2ms    (main thread, single frame)
    Stage 3: N Ticks  (N = ceil(label_count / 18))
    Stage 4: < 1ms    (handle release, deferred GPU eviction)

## SECTION 4 — HARDWARE & SCALABILITY

### [SCALE-LOC-001] VRAM Optimization — 2K Atlas Packing, No 4K on MX350
All language glyph sets MUST be packed into 2048×2048 atlases maximum.
4K atlases (4096×4096) are FORBIDDEN on hardware profiles including or below MX350.

public static class AtlasPolicy
{
    public static int GetMaxAtlasSize(GpuProfile profile)
    {
        return profile switch
        {
            GpuProfile.MX350_Or_Below  => 2048,  // HARD LIMIT — [SCALE-LOC-001]
            GpuProfile.GTX1060_Class   => 2048,  // Conservative — share VRAM with scene
            GpuProfile.RTX3060_And_Up  => 4096,  // Permitted only for CJK full glyph set
            _                          => 2048   // Default safe
        };
    }
}

Glyph Packing Strategy:

ATLAS LAYOUT — CJK + Latin + Cyrillic (2K, ~1.8MB VRAM)
[STRUCT_START]
 [0,0][2048,0]
  Latin Extended + Cyrillic   Arabic/Hebrew RTL
  (512×2048 column)           (256×2048 column)

  CJK Unified (BMP subset)    Hangul Syllables
  (1024×1024 block)           (256×512 block)

  Numeric/Symbol shared zone  [RESERVED]
  (all languages)             Glitch variants
[0,2048][2048,2048]
[STRUCT_END]

PACKING RULE: CJK subset = 2500 most-frequent ideographs only.
Full CJK (20,000+) requires RTX3060+ profile AND player opt-in.

### [SCALE-LOC-002] Dynamic Atlas Scope — User Input Only
TMP Dynamic Atlas mode introduces per-glyph GPU texture uploads. This is acceptable
ONLY for unpredictable user-generated content (chat, callsign input).

| Context              | Atlas Mode     | Justification                              |
|----------------------|----------------|--------------------------------------------|
| Story/Narrative text | Static-Baked   | Glyph set known at build time              |
| HUD labels           | Static-Baked   | Fixed key set, pre-resolved                |
| World/NPC names      | Static-Baked   | Authored strings, finite set               |
| Chat / Callsign      | Dynamic    | Arbitrary user input, unknown glyph set    |
| Debug console        | Dynamic        | Dev-only, excluded from ship builds        |

// Enforced at TMP_TextRegistry registration time
public void Register(TMP_Text text, LocLayer layer, bool isUserInput = false)
{
    if (!isUserInput && text.font.atlasPopulationMode == AtlasPopulationMode.Dynamic)
    {
        // [SCALE-LOC-002] Violation — story/HUD text must use static atlas
        Debug.LogError(
            $"[LOC-BABEL] Dynamic atlas on non-user-input text '{text.name}'. " +
            $"Switch to Static-Baked. Layer: {layer}",
            text.gameObject
        );
        // Force correct mode at runtime to prevent VRAM leaks
        text.font.atlasPopulationMode = AtlasPopulationMode.Static;
    }
    _registry.Add(new TMP_TextEntry(text, layer, isUserInput));
}

## SECTION 5 — INTEGRATION & FAILURE MODES

### [SAFE-LOC-001] Missing Key Fallback — Log-Once, No Exception
If a key hash resolves to no entry across all three layers (Core, World, Narrative),
the system MUST return the literal string `[ERR_MISSING_KEY]` and log exactly once per
missing key per session. Exceptions MUST NOT be thrown.

public static class LocErrorLog
{
    // HashSet — tracks already-logged missing keys to enforce log-once
    private static readonly HashSet<int> _loggedMissing = new HashSet<int>(64);

    public static void LogOnce(int keyHash)
    {
        if (_loggedMissing.Add(keyHash)) // Add returns false if already present
        {
            // Log with full context for diagnostics
            Debug.LogWarning(
                $"[LOC-BABEL] Missing localization key. Hash: 0x{keyHash:X8} | " +
                $"Language: {LocalizationCoordinator.ActiveLanguage} | " +
                $"Frame: {Time.frameCount}"
            );
            // Optional: telemetry ping (non-blocking, fire-and-forget)
            Telemetry.RecordOnce("loc_missing_key", keyHash);
        }
        // Second+ call for same key: silent. No log. No exception.
    }
}

// Contract: this span is stored as a static literal — zero alloc
private static class MissingKeyLiteral
{
    private static readonly char[] _chars = "[ERR_MISSING_KEY]".ToCharArray();
    public static ReadOnlySpan<char> AsSpan() => _chars.AsSpan();
}

### [SAFE-LOC-002] Overflow Auto-Scale — Matrix-Based, Range [0.8, 1.0]
When a localized string overflows its assigned UI bounds (common in German, Finnish, Arabic
expanded forms), the system MUST auto-scale text using Matrix4x4 transform only.
`rectTransform.localScale` is FORBIDDEN for this purpose (cascades to child layouts).

public static class LocOverflowHandler
{
    private const float SCALE_MIN = 0.8f;
    private const float SCALE_MAX = 1.0f;

    /// <summary>
    /// Applies matrix-based scale to fit text within bounds.
    /// Called after text assignment, before next render frame.
    /// </summary>
    public static void FitToBounds(TMP_Text text, RectTransform bounds)
    {
        text.ForceMeshUpdate(ignoreActiveState: true);

        float textWidth    = text.preferredWidth;
        float boundsWidth  = bounds.rect.width;

        if (textWidth <= boundsWidth)
        {
            // No overflow — reset to identity
            text.rectTransform.localMatrix = Matrix4x4.identity;
            return;
        }

        // Compute required scale — clamped to [0.8, 1.0]
        float rawScale   = boundsWidth / textWidth;
        float safeScale  = Mathf.Clamp(rawScale, SCALE_MIN, SCALE_MAX);

        // Apply via matrix — does NOT affect child layout transforms
        Matrix4x4 scaleMatrix = Matrix4x4.Scale(new Vector3(safeScale, safeScale, 1f));
        text.rectTransform.localMatrix = scaleMatrix;

        // If even SCALE_MIN is insufficient, truncate with ellipsis
        if (rawScale < SCALE_MIN)
        {
            text.overflowMode   = TextOverflowModes.Ellipsis;
            text.enableWordWrapping = false;
            // Log for loc team — string needs shortening in source
            Debug.LogWarning(
                $"[LOC-BABEL] Overflow unresolvable at 0.8x scale. " +
                $"Key: 0x{text.GetHashCode():X8} | Lang: {LocalizationCoordinator.ActiveLanguage}"
            );
        }
    }
}

## SECTION 6 — REGISTRY SPECIFICATION

### [SPEC-LOC-001] TMP_TextRegistry — Mandatory Registration Contract

/// <summary>
/// Central registry for all TMP_Text instances in HECTON-8.
/// Registration is MANDATORY in Awake for every TMP_Text component.
/// FindObjectsOfType<TMP_Text> is FORBIDDEN — [FORBID-LOC-002].
/// </summary>
public sealed class TMP_TextRegistry : MonoBehaviour
{
    private static TMP_TextRegistry _instance;

    // Layered storage — index by LocLayer for O(1) layer enumeration
    private readonly List<TMP_TextEntry>[] _layers
        = new List<TMP_TextEntry>[3]
          {
              new List<TMP_TextEntry>(128),  // Core
              new List<TMP_TextEntry>(512),  // World
              new List<TMP_TextEntry>(1024)  // Narrative
          };

    public static void Register(TMP_Text text, LocLayer layer, bool isUserInput = false)
    {
        if (_instance == null)
        {
            Debug.LogError("[LOC-BABEL] TMP_TextRegistry not initialized.");
            return;
        }
        _instance.RegisterInternal(text, layer, isUserInput);
    }

    public static void Unregister(TMP_Text text)
        => _instance?.UnregisterInternal(text);

    public static void ForEachInLayer(LocLayer layer, Action<TMP_TextEntry> action)
    {
        var list = _instance._layers[(int)layer];
        for (int i = 0; i < list.Count; i++)
        {
            if (list[i].Text != null) // Guard: destroyed objects
                action(list[i]);
        }
    }

    public static void ForEachAll(Action<TMP_TextEntry> action)
    {
        foreach (var layer in _instance._layers)
            for (int i = 0; i < layer.Count; i++)
                if (layer[i].Text != null)
                    action(layer[i]);
    }

    private void RegisterInternal(TMP_Text text, LocLayer layer, bool isUserInput)
    {
        // Validate atlas mode — [SCALE-LOC-002]
        if (!isUserInput && text.font?.atlasPopulationMode == AtlasPopulationMode.Dynamic)
        {
            Debug.LogError($"[LOC-BABEL] Dynamic atlas violation on '{text.name}'.", text);
            text.font.atlasPopulationMode = AtlasPopulationMode.Static;
        }
        _layers[(int)layer].Add(new TMP_TextEntry(text, layer, isUserInput));
    }

    private void UnregisterInternal(TMP_Text text)
    {
        foreach (var layer in _layers)
            for (int i = layer.Count - 1; i >= 0; i--)
                if (layer[i].Text == text) { layer.RemoveAt(i); break; }
    }
}

public readonly struct TMP_TextEntry
{
    public readonly TMP_Text  Text;
    public readonly LocLayer  Layer;
    public readonly int       KeyHash;
    public readonly bool      IsUserInput;

    public TMP_TextEntry(TMP_Text text, LocLayer layer, bool isUserInput)
    {
        Text        = text;
        Layer       = layer;
        KeyHash     = LocHash.Compute(text.name); // default; override via SetKey()
        IsUserInput = isUserInput;
    }
}
