### NEIGHBOR_QUERY

get_neighbors(pos, radius):
  min_cell = cell_index(pos - radius)
  max_cell = cell_index(pos + radius)
  neighbors = []
  for x in [min_cell.x .. max_cell.x]:
    for y in [min_cell.y .. max_cell.y]:
      for z in [min_cell.z .. max_cell.z]:
        cell_idx = x + y*32 + z*1024
        count = min(grid_counter[cell_idx], 32)
        for i in [0..count):
          candidate = grid_cells[cell_idx * 32 + i]
          if length(boid[candidate].pos - pos) < radius:
            neighbors.push(candidate)
            if neighbors.size >= 32: return neighbors
  return neighbors

## FLOCKING_KERNEL

### BOID_DATA_STRUCT

struct BoidData {
  float3 pos;
  float3 vel;
  uint state;        // bits [0-1]=behavior, [2-7]=health, [8-15]=target_id
  float energy;
}

### STATE_DECODE

behavior   = state & 0x3
health     = (state >> 2) & 0x3F
target_id  = (state >> 8) & 0xFF

### FLOCKING_RULES

perception_radius = 8.0
separation_radius = 2.5

[numthreads(256,1,1)]
KernelFlocking(id):
  if (state & 0x3) == 3: return  // consumed

  neighbors = get_neighbors(pos[id], perception_radius)

  // Separation
  sep_force = float3(0,0,0)
  for n in neighbors:
    diff = pos[id] - pos[n]
    dist = max(length(diff), 0.01)
    if dist < separation_radius:
      sep_force += diff / (dist * dist)

  // Alignment
  avg_vel = float3(0,0,0)
  for n in neighbors:
    avg_vel += vel[n]
  avg_vel /= max(neighbors.size, 1)
  align_force = avg_vel - vel[id]

  // Cohesion
  center = float3(0,0,0)
  for n in neighbors:
    center += pos[n]
  center /= max(neighbors.size, 1)
  cohesion_force = center - pos[id]

  // Weights by state
  w_sep = (behavior == 1) ? 3.5 : 1.8
  w_ali = (behavior == 1) ? 0.2 : 1.0
  w_coh = (behavior == 1) ? 0.1 : 0.8

  accel = sep_force * w_sep + align_force * w_ali + cohesion_force * w_coh

  // Clamp
  if length(accel) > max_force:
    accel = normalize(accel) * max_force

  vel[id] += accel * dt

  // Speed limits
  speed = length(vel[id])
  max_speed = (behavior == 1) ? 12.0 : 6.0
  if speed > max_speed:
    vel[id] = (vel[id] / speed) * max_speed

  pos[id] += vel[id] * dt

## POSITION_BASED_DYNAMICS

### SPHERE_COLLISION_CONSTRAINT

[numthreads(256,1,1)]
KernelPBDSolve(id):
  if (state & 0x3) == 3: return

  neighbors = get_neighbors(pos[id], 2.0)

  for n in neighbors:
    if n == id: continue
    diff = pos[id] - pos[n]
    dist = length(diff)
    min_dist = 1.2  // boid radius * 2

    if dist < min_dist && dist > 0.001:
      dir = diff / dist
      correction = dir * (min_dist - dist) * 0.5
      pos[id] += correction
      pos[n] -= correction

### PANIC_TRIGGER

threat_distance_sq = 2500  // 50m squared

[numthreads(256,1,1)]
KernelUpdateBehavior(id):
  dist_sq = length_sq(leviathan_pos - pos[id])

  current_behavior = state[id] & 0x3
  target_behavior = (dist_sq < threat_distance_sq) ? 1 : 0

  // Exponential blend
  k = 4.0
  blend = 1.0 - exp(-k * dt)

  if current_behavior != target_behavior:
    state[id] = (state[id] & ~0x3) | target_behavior
    vel[id] += normalize(pos[id] - leviathan_pos) * 8.0 * blend

## NAN_GUARDS

### SAFE_NORMALIZE

safe_normalize(v):
  len_sq = dot(v, v)
  if len_sq < 1e-6:
    return float3(0, 1, 0)  // default up
  return v * rsqrt(len_sq)
