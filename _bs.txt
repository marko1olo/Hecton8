3|// Canonical bootstrap read-model for runtime consumers.
4|//
5|// GameBootstrapper remains the owner and publisher. Runtime systems read this
6|// state to avoid depending on the full bootstrap owner when they only need
7|// lifecycle/player facts.
8|// ============================================================================

10|using UnityEngine;
11|
12|namespace Hecton8.Core
13|{
14|    /// <summary>
15|    /// Canonical runtime read-model for bootstrap lifecycle and current player reference.
16|    /// </summary>
17|    public static class BootstrapState
18|    {
19|        /// <summary>
20|        /// Bootstrap completed initialization, player activation, and world-ready handoff.
21|        /// </summary>
22|        public static bool IsGameReady { get; private set; }
23|
24|        /// <summary>
25|        /// A live GameBootstrapper instance currently owns scene startup.
26|        /// </summary>
27|        public static bool HasActiveInstance { get; private set; }
28|
29|        /// <summary>
30|        /// Last known player GameObject published by bootstrap.
31|        /// </summary>
32|        public static GameObject CurrentPlayerObject { get; private set; }
33|
34|        /// <summary>
35|        /// Fast-access player transform for runtime systems that must avoid scene search.
36|        /// </summary>
37|        public static Transform CurrentPlayerTransform =>
38|            CurrentPlayerObject != null ? CurrentPlayerObject.transform : null;
39|
40|        /// <summary>
41|        /// Publishes whether bootstrap currently owns runtime startup.
42|        /// </summary>
43|        /// <param name="hasActiveInstance">True while GameBootstrapper owns scene activation.</param>
44|        public static void PublishBootstrapPresence(bool hasActiveInstance)
45|        {
46|            HasActiveInstance = hasActiveInstance;
47|        }
48|
49|        /// <summary>
50|        /// Publishes the world-ready handoff state.
51|        /// </summary>
52|        /// <param name="isGameReady">True after bootstrap completes and gameplay may proceed.</param>
53|        public static void PublishGameReady(bool isGameReady)
54|        {
55|            IsGameReady = isGameReady;
56|        }
57|
58|        /// <summary>
59|        /// Publishes the current player reference for runtime consumers.
60|        /// </summary>
61|        /// <param name="playerObject">Resolved player object, or null.</param>
62|        public static void PublishCurrentPlayerObject(GameObject playerObject)
63|        {
64|            CurrentPlayerObject = IsProductionPlayerAuthorityObject(playerObject) ? playerObject : null;
65|        }
66|
67|        /// <summary>
68|        /// Clears the cached player reference if it still belongs to the specified object.
69|        /// </summary>
70|        /// <param name="playerObject">Player object being invalidated.</param>
71|        public static void ClearCurrentPlayerObject(GameObject playerObject)
72|        {
73|            if (ReferenceEquals(CurrentPlayerObject, playerObject))
74|                CurrentPlayerObject = null;
75|        }
76|
77|        /// <summary>
78|        /// Tries to resolve the current player transform without scene search.
79|        /// </summary>
80|        /// <param name="playerTransform">Resolved transform when available.</param>
81|        /// <returns>True when a current player transform exists.</returns>
82|        public static bool TryGetCurrentPlayerTransform(out Transform playerTransform)
83|        {
84|            playerTransform = CurrentPlayerTransform;
85|            return playerTransform != null;
86|        }
87|
88|        /// <summary>
89|        /// Resets published bootstrap state during domain reload or play mode exit.
90|        /// </summary>
91|        public static void Reset()
92|        {
93|            IsGameReady = false;
94|            HasActiveInstance = false;
95|            CurrentPlayerObject = null;
96|        }
97|

99|        /// Validates that the object owns production player authority without referencing gameplay assemblies.
