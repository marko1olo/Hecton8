140|                    $"[GlobalRegistry] Unregister called for non-registered {typeof(T).Name}.");
141|                _unregisterMissLogged = true;
142|            }
143|#endif
144|            return false;
145|        }

147|        /// <summary>
148|        /// Removes an item without emitting a miss warning; used by idempotent lifecycle teardown paths.
149|        /// </summary>
150|        /// <param name="item">Item instance to remove.</param>
151|        /// <returns>True when the item was found and removed.</returns>
152|        public bool TryUnregister(T item)
153|        {
154|            for (int i = 0; i < _count; i++)
155|            {
156|                if (!ReferenceEquals(_items[i], item))
157|                    continue;
158|
159|                _count--;
160|                _items[i] = _items[_count];
161|                _items[_count] = null;
162|#if UNITY_EDITOR || DEVELOPMENT_BUILD
163|                _indexErrorLogged = false;
164|                _unregisterMissLogged = false;
165|                _destroyedEntryLogged = false;
166|#endif
167|                return true;
168|            }
169|
170|            return false;
171|        }

173|        /// <summary>
174|        /// Checks whether a given instance is already present in the bucket.
175|        /// </summary>
176|        /// <param name="item">Item instance to test.</param>
177|        /// <returns>True when the instance is present.</returns>
178|        public bool Contains(T item)
179|        {
180|            for (int i = 0; i < _count; i++)
181|            {
182|                if (ReferenceEquals(_items[i], item))
183|                    return true;
184|            }
185|
186|            return false;
187|        }

189|        /// <summary>
190|        /// Clears all live entries and nulls the dense storage window.
191|        /// </summary>
192|        public void Clear()
193|        {
194|            Array.Clear(_items, 0, _count);
195|            _count = 0;
196|#if UNITY_EDITOR || DEVELOPMENT_BUILD
197|            _indexErrorLogged = false;
198|            _nullRegistrationLogged = false;
199|            _capacityErrorLogged = false;
200|            _unregisterMissLogged = false;
201|            _destroyedEntryLogged = false;
202|#endif
203|        }
