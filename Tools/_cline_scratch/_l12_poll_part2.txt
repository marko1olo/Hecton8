def proc_alive(pid):
    if not pid: return False
    if ctypes is not None:
        try:
            k = ctypes.windll.kernel32
            h = k.OpenProcess(0x1000, False, int(pid))
            if h:
                code = ctypes.c_ulong()
                ok = k.GetExitCodeProcess(h, ctypes.byref(code))
                k.CloseHandle(h)
                if ok: return code.value == 259
        except Exception:
            pass
    try:
        import subprocess
        p = subprocess.run(["tasklist", "/FI", "PID eq %s" % pid, "/NH"], capture_output=True, text=True)
        return str(pid) in (p.stdout or "")
    except Exception:
        return False
def read_pid():
    if not os.path.isfile(PIDF): return None
    try: return int(open(PIDF, encoding="utf-8").read().strip())
    except Exception: return None
def load_prev_size():
    if not os.path.isfile(STATE): return None
    try:
        import json
        return json.load(open(STATE, encoding="utf-8")).get("log_bytes")
    except Exception: return None
def save_state(log_bytes, pid, alive):
    try:
        import json
        json.dump({"ts": time.strftime("%Y-%m-%d %H:%M:%S"), "log_bytes": log_bytes, "pid": pid, "alive": alive}, open(STATE, "w", encoding="utf-8"), indent=2)
    except Exception: pass
