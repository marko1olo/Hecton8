157|            ref NativeArray<byte> payload,
158|            string owner,
159|            string label,
160|            Allocator allocator = Allocator.Temp)
161|        {
162|            if (!payload.IsCreated)
163|                return;

165|            IntPtr payloadPointer = ResolveTransientPayloadPointer(payload);
166|            int registrationId = TryGetTransientPayloadRegistrationId(payloadPointer);
167|            if (registrationId <= 0)

169|                payload.Dispose();

171|                throw new InvalidOperationException(TransientPayloadUnregistrationFailureMessage);
172|            }
173|
174|            try
175|            {
176|                payload.Dispose();
177|                payload = default;
178|            }
179|            catch (Exception disposalException)
180|            {

183|                    throw new AggregateException(TransientPayloadUnregistrationFailureMessage, disposalException);
184|                }
185|
186|                throw;
187|            }
188|
189|            bool unregistered = TryUnregisterTransientNativeArrayPayload(payloadPointer, registrationId);
190|            if (!unregistered)

192|        }
