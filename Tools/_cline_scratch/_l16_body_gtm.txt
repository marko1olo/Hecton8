  392:         public void Tick(float deltaTime)
  393:         {
  394:             if (!AreTickListsReady())
  395:                 return;
  396: 
  397:             float dt = deltaTime;
  398:             float slowTickDt = dt;
  399:             if (_runtimeGameplayBootstrapGateActive &&
  400:                 BootstrapState.HasActiveInstance &&
  401:                 !BootstrapState.IsGameReady &&
  402:                 slowTickDt <= 0f)
  403:             {
  404:                 slowTickDt = SystemDispatcher.CurrentFrameUnscaledDeltaTime;
  405:                 if (slowTickDt <= 0f)
  406:                     slowTickDt = 0.0166667f;
  407:             }
  408: #if UNITY_EDITOR || DEVELOPMENT_BUILD
  409:             if (!_loggedFirstUpdateExecution && enableSlowTickProfiling)
  410:                 _loggedFirstUpdateExecution = true;
  411: #endif
  412: 
  413:             _tickables.BeginIteration();
  414:             try
  415:             {
  416:                 int count = _tickables.Count;
  417: 
  418:                 for (int i = 0; i < count; i++)
  419:                 {
  420:                     ITickable item = _tickables.GetAt(i);
  421: 
  422:                     if (item == null)
  423:                     {
  424:                         _tickables.Remove(item);
  425:                         continue;
  426:                     }
  427: 
  428:                 // ── Auto-Cleanup: "fake null" (unichtozhennyy MonoBehaviour) ──
  429:                     if (item is UnityEngine.Object obj && obj == null)
  430:                     {
  431:                         _tickables.Remove(item);
  432:                         continue;
  433:                     }
  434: 
  435:                     item.Tick(dt);
  436:                 }
  437:             }
  438:             finally
  439:             {
  440:                 _tickables.EndIteration();
  441:             }
  442:             ProcessSlowTickIfNeeded(slowTickDt);
  443: 
  444: #if UNITY_EDITOR
  445:             _debugTickCount = _tickables.Count;
  446: #endif
  447:         }
  448:
