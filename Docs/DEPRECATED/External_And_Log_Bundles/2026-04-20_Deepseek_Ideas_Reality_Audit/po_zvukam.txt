import numpy as np
from scipy.io import wavfile

# --- [ KONFIGURATsIYa ] ---
SAMPLE_RATE = 44100
DURATION = 4.5  # sekund. Idealno dlya narastayuschego saspensa.
t = np.linspace(0, DURATION, int(SAMPLE_RATE * DURATION), endpoint=False)
NUM_SAMPLES = len(t)

print("[1/6] Initsializatsiya generatora davleniya...")

# --- [ LAYER 1: SUB-BASS PRESSURE (Dyhanie bezdny) ] ---
# Generiruem Belyy shum i prevraschaem ego v Korichnevyy shum (Brown Noise) cherez FFT-filtr (1/f^2)
white_noise = np.random.normal(0, 1, NUM_SAMPLES)
W_freq = np.fft.rfft(white_noise)
freqs = np.fft.rfftfreq(NUM_SAMPLES, 1.0 / SAMPLE_RATE)
freqs[0] = 1.0  # Izbegaem deleniya na nol

# Filtr: zatuhanie vysokih chastot. Otrezaem vse, chto nizhe 20 Hz, chtoby ne spalit sabvufery igrokov.
filter_curve = 1.0 / (freqs ** 1.2)
filter_curve[freqs < 20.0] = 0.0 
W_freq *= filter_curve

brown_noise = np.fft.irfft(W_freq, n=NUM_SAMPLES)
brown_noise /= np.max(np.abs(brown_noise))  # Normalizatsiya

# LFO: medlennoe, udushayuschee pulsirovanie (0.3 Gts)
lfo_pressure = 0.6 + 0.4 * np.sin(2 * np.pi * 0.3 * t)
layer1_sub = brown_noise * lfo_pressure

print("[2/6] Sintez metallicheskogo skrezheta (FM-Synthesis)...")

# ---[ LAYER 2: METALLIC FRICTION (Skrezhet korpusa) ] ---
# Additivnyy FM-sintez: negarmonicheskie chastoty, imitiruyuschie tolstye titanovye plity
f_carriers =[120.0, 315.0, 542.0, 830.0]
f_modulators =[45.0, 87.0, 133.0, 311.0]

# Ogibayuschaya osnovnogo stona: dva narastaniya davleniya za 4.5 sekundy
groan_env = np.sin(2 * np.pi * 0.22 * t) ** 4 

# Effekt "Stick-Slip": preryvistoe trenie metalla
chatter_noise = np.random.uniform(0.5, 1.0, NUM_SAMPLES)
stick_slip = 0.7 + 0.3 * np.sin(2 * np.pi * 35.0 * t) * chatter_noise

layer2_metal = np.zeros(NUM_SAMPLES)

for fc, fm in zip(f_carriers, f_modulators):
    # Metall gnetsya: chastota medlenno polzet vniz pod davleniem
    pitch_slide = 1.0 - 0.15 * np.linspace(0, 1, NUM_SAMPLES)
    
    # Integriruem fazu dlya korrektnogo izmeneniya pitcha (Pitch Bend)
    phase_c = np.cumsum(fc * pitch_slide) / SAMPLE_RATE * 2 * np.pi
    phase_m = np.cumsum(fm * pitch_slide) / SAMPLE_RATE * 2 * np.pi
    
    # Indeks modulyatsii: chem silnee davlenie (groan_env), tem agressivnee garmoniki
    mod_index = 6.0 * groan_env * stick_slip
    
    modulator = np.sin(phase_m)
    carrier = np.sin(phase_c + mod_index * modulator)
    
    layer2_metal += carrier * groan_env

# Normalizuem sloy skrezheta
layer2_metal /= np.max(np.abs(layer2_metal))

print("[3/6] Proschet mikrotreschin i lopayuschihsya zaklepok...")

# --- [ LAYER 3: MICRO-FRACTURES (Tranzienty) ] ---
# Generiruem 30 sluchaynyh tochek "treska"
layer3_cracks = np.zeros(NUM_SAMPLES)
num_pops = 30
pop_times = np.random.randint(0, NUM_SAMPLES - 1000, num_pops)

# Sozdaem rezkie ekspansivnye vspleski shuma
crack_noise = np.random.normal(0, 1, NUM_SAMPLES)
crack_env = np.zeros(NUM_SAMPLES)

for p in pop_times:
    # Ostryy eksponentsialnyy spad
    decay = np.exp(-np.linspace(0, 15, 1000))
    crack_env[p:p+1000] += decay * np.random.uniform(0.5, 1.0)

layer3_cracks = crack_noise * crack_env

# High-Pass filtr (matematicheskaya proizvodnaya ot signala 2-go poryadka)
# Otsekaem vse nizy, ostavlyaem tolko ostryy "hrust" peska/metalla
layer3_cracks = np.diff(layer3_cracks, prepend=0)
layer3_cracks = np.diff(layer3_cracks, prepend=0)
layer3_cracks /= (np.max(np.abs(layer3_cracks)) + 1e-9)

print("[4/6] Mikshirovanie i Soft Clipping (Iskazhenie)...")

# --- [ SVEDENIE I SOFT CLIPPING ] ---
# Proportsii miksa
MIX_SUB = 0.85
MIX_METAL = 0.70
MIX_CRACKS = 0.35

mixed = (layer1_sub * MIX_SUB) + (layer2_metal * MIX_METAL) + (layer3_cracks * MIX_CRACKS)

# Imitiruem peregruz akusticheskoy sredy (Soft Clipping) cherez giperbolicheskiy tangens.
# Drive = 2.5 splyuschivaet piki, delaet zvuk gryaznym i monolitnym.
DRIVE = 2.5
distorted = np.tanh(mixed * DRIVE)

print("[5/6] Obrabotka reverberatsiey kabiny skutera...")

# ---[ EFFEKTY: SMALL CABIN DELAY / REVERB ] ---
# Primitivnaya liniya zaderzhki. Ispolzuem prostye chisla millisekund dlya izbezhaniya rezonansov.
delays_ms =[17, 23, 37, 43]
decays =[0.4, 0.3, 0.2, 0.15]

reverbed = np.copy(distorted)
for ms, decay in zip(delays_ms, decays):
    shift_samples = int((ms / 1000.0) * SAMPLE_RATE)
    shifted_signal = np.zeros(NUM_SAMPLES)
    shifted_signal[shift_samples:] = distorted[:-shift_samples] * decay
    reverbed += shifted_signal

print("[6/6] Mastering i eksport...")

# ---[ MASTERING I EKSPORT ] ---
# Zaschita ot klippinga i privedenie k 16-bit
final_audio = reverbed / np.max(np.abs(reverbed)) * 0.95 
audio_16bit = np.int16(final_audio * 32767)

wavfile.write("abyssal_hull_groan.wav", SAMPLE_RATE, audio_16bit)
print("Gotovo! Fayl 'abyssal_hull_groan.wav' sgenerirovan uspeshno.")

Phase Accumulation (Integrirovanie fazy): V bloke FM-sinteza chastota (Pitch Slide) integriruetsya cherez np.cumsum. Eto daet fizicheski-realistichnoe i gladkoe «spolzanie» pitcha metalla pod nagruzkoy bez artefaktov (v otlichie ot umnozheniya vremeni na izmenyaemuyu chastotu napryamuyu).

Stick-Slip Effect: Nizkochastotnaya shumovaya modulyatsiya (chatter_noise) zastavlyaet FM-indeks dergatsya. Iz-za etogo skrezhet zvuchit ne kak rovnyy sint, a kak metall, kotoryy s trudom polzet ryvkami drug po drugu.

High-Pass (Proizvodnye): Ispolzovanie dvukratnogo np.diff() daet spad 12 dB/oktavu snizu vverh. Eto mgnovennyy sposob v chistom NumPy srezat ves niz iz impulsov zaklepok, ostaviv tolko pugayuschiy, vysokochastotnyy "pesok".

Soft Clipping (tanh): np.tanh — klassika tsifrovogo overdrayva. Szhimaet dinamicheskiy diapazon kak lampovyy usilitel, "skleivaya" gul, skrezhet i schelchki v odno akusticheskoe sobytie.

import numpy as np
from scipy.io import wavfile

def generate_sonar_ping():
    # Bazovye parametry
    sample_rate = 44100
    duration = 4.0  # Obschaya dlitelnost zvuka s uchetom zatuhaniya eha
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)

    print("Initsializatsiya akusticheskih podsistem HECTON-8...")

    # ==========================================
    # SLOY 1: THE ATTACK (Pezo-udar)
    # ==========================================
    # Ochen rezkiy schelchok. My generiruem vysokochastotnyy sinus i belyy shum,
    # a zatem "dushim" ih sverhbystroy eksponentsialnoy ogibayuschey (~5ms).
    attack_duration = 0.1
    t_attack = t[:int(sample_rate * attack_duration)]
    
    # Ogibayuschaya: np.exp(-t * 1000) padaet pochti do nulya kak raz k 5-y millisekunde
    env_attack = np.exp(-t_attack * 1000) 
    
    attack_sine = np.sin(2 * np.pi * 4500 * t_attack) # Rezkiy rezonans pezokristalla
    attack_noise = np.random.randn(len(t_attack))     # Metallicheskiy lyazg/iskra
    
    attack_signal = env_attack * (attack_sine + 1.2 * attack_noise)
    attack_full = np.pad(attack_signal, (0, len(t) - len(attack_signal)))

    # ==========================================
    # SLOY 2: THE CHIRP (Osnovnoy probivayuschiy impuls)
    # ==========================================
    # Lineynaya chastotnaya modulyatsiya (LFM) ot 2000 Gts do 400 Gts za 0.5s.
    chirp_duration = 0.5
    t_chirp = t[:int(sample_rate * chirp_duration)]
    f_start = 2000
    f_end = 400
    
    # Integral ot lineynogo izmeneniya chastoty daet kvadratichnuyu fazu
    phase_chirp = 2 * np.pi * (f_start * t_chirp + ((f_end - f_start) / (2 * chirp_duration)) * t_chirp**2)
    
    # Ogibayuschaya chirpa: moschnyy start s plavnym zatuhaniem k kontsu 0.5s
    env_chirp = np.exp(-t_chirp * 5)
    chirp_signal = env_chirp * np.sin(phase_chirp)
    chirp_full = np.pad(chirp_signal, (0, len(t) - len(chirp_signal)))

    # ==========================================
    # SLOY 3: THE ABYSSAL TAIL (Glubinnoe eho)
    # ==========================================
    # Gryaznyy nizkochastotnyy rokot s "effektom bieniy" (raznitsa chastot 150 i 147 Gts)
    tail_carrier = (
        np.sin(2 * np.pi * 150 * t) + 
        0.6 * np.sin(2 * np.pi * 147 * t) +  # Rasstroyka sozdaet pulsatsiyu tolschi vody
        0.4 * np.sin(2 * np.pi * 300 * t)
    )

    # Generatsiya nizkochastotnogo moduliruyuschego shuma (LFO)
    # Sozdaem 15 sluchaynyh tochek i plavno interpoliruem ih na vsyu dlinu (sozdaet mutnye "volny")
    num_lfo_points = int(duration * 4) 
    lfo_noise = np.interp(
        np.linspace(0, 1, len(t)), 
        np.linspace(0, 1, num_lfo_points), 
        np.random.uniform(0.1, 1.0, num_lfo_points)
    )

    # Dobavlyaem vysokochastotnyy "pesok/il" — sglazhennyy belyy shum
    mud_noise = np.convolve(np.random.randn(len(t)), np.ones(40)/40, mode='same')

    # Ogibayuschaya hvosta: eho nachinaet narastat s 0.1s (srazu posle udara), 
    # dostigaet pika i dolgo zatuhaet v bezdne (4 km)
    env_tail = np.clip((t - 0.1) / 0.3, 0, 1) * np.exp(-(t - 0.1) * 0.9)

    tail_full = env_tail * lfo_noise * (tail_carrier + 0.3 * mud_noise)

    # ==========================================
    # MIKSING I MASTERING
    # ==========================================
    print("Kompilyatsiya zvukovyh sloev i saturatsiya...")
    
    # Smeshivaem sloi s nuzhnymi proportsiyami
    mixed_signal = 0.9 * attack_full + 1.0 * chirp_full + 0.7 * tail_full

    # Saturatsiya (Soft-clipping cherez giperbolicheskiy tangens)
    # Eto "splyuschivaet" piki signala, dobavlyaya emu harakternuyu elektromagnitnuyu teplotu i mosch
    driven_signal = np.tanh(mixed_signal * 1.8) / 1.8

    # Normalizatsiya v 16-bitnyy PCM
    final_signal = driven_signal / np.max(np.abs(driven_signal))
    audio_data = np.int16(final_signal * 32767)

    # Eksport
    output_filename = "sonar_active.wav"
    wavfile.write(output_filename, sample_rate, audio_data)
    print(f"Gotovo! Akusticheskiy profil sohranen kak: {output_filename}")

if __name__ == "__main__":
    generate_sonar_ping()
Razbor mehaniki zvuka (pochemu on zvuchit imenno tak):

Saturatsiya (np.tanh): Eto sekretnyy ingredient. Vmesto chistogo matematicheskogo slozheniya, giperbolicheskiy tangens imitiruet peregruzku analogovyh tsepey. Udar i nachalo LFM-chirpa "sminayutsya", sozdavaya effekt kolossalnogo rashoda energii generatorom skutera.

Phase Calculation: Chirp napisan ne cherez vstroennye funktsii scipy.signal, a chisto matematicheski cherez integral chastoty. Eto daet idealnyy, nepreryvnyy spad ot 2000 Gts do 400 Gts bez "schelchkov" po faze.

Beating Effect (Bieniya): V glubine hvosta zvuchat sinusoidy na 150 Gts i 147 Gts. Raznitsa v 3 Gts zastavlyaet zvuk pulsirovat (3 raza v sekundu), chto psihoakusticheski vosprinimaetsya kak ogromnoe, zamknutoe vodnoe prostranstvo ili otrazhenie ot nerovnyh sten kanona.

LFO Interpolyatsiya: My generiruem sluchaynyy massiv dannyh ochen nizkoy chastoty i natyagivaem ego (np.interp) na vsyu dlinu audio. Eto zastavlyaet reverberatsiyu (hvost) "dyshat" i perekatyvatsya, imitiruya nerovnosti relefa (te samye vokselnye skaly).

import numpy as np
from scipy.io import wavfile

def generate_thruster_loop(duration=5.0, load=0.85):
    """
    Generiruet besshovnyy zvuk dvigatelya Manta.
    :param duration: Dlitelnost lupa (strogo 5.0 dlya sovpadeniya faz).
    :param load: Nagruzka na dvigatel (0.0 - 1.0). Chem vyshe, tem zhestche kavitatsiya.
    """
    sample_rate = 44100
    # endpoint=False vazhen dlya besshovnogo zatsiklivaniya, chtoby posledniy sempl ne dubliroval pervyy
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print(f"Zapusk reaktora Manta... Nagruzka: {load*100}%")

    # ==========================================
    # SLOY 1: THE HUM (Elektromotor)
    # ==========================================
    # Bazovaya chastota 80 Gts (idealno kratna 5 sekundam: 80 * 5 = 400 polnyh tsiklov)
    # Garmoniki sozdayut "napryazhennoe" zhuzhzhanie obmotok statora
    hum = (
        1.00 * np.sin(2 * np.pi * 80 * t) + 
        0.60 * np.sin(2 * np.pi * 160 * t) + 
        0.35 * np.sin(2 * np.pi * 240 * t) + 
        0.15 * np.sin(2 * np.pi * 320 * t)
    )

    # ==========================================
    # PODGOTOVKA IDEALNOGO ShUMA (FFT-metod)
    # ==========================================
    # Generiruem massiv chastot. Ispolzovanie IFFT garantiruet, 
    # chto sgenerirovannyy shum budet besshovnym pri zatsiklivanii.
    freqs = np.fft.rfftfreq(N, 1 / sample_rate)
    
    # ==========================================
    # SLOY 2: THE FLOW (Turbulentnost vody)
    # ==========================================
    # Sozdaem Rozovyy shum (spad energii 1/sqrt(f))
    pink_mag = np.zeros_like(freqs)
    pink_mag[1:] = 1.0 / np.sqrt(freqs[1:]) 
    
    # Low-pass filtr: propuskaem vse do 600 Gts, zatem plavnyy spad
    lp_filter = np.where(freqs <= 600, 1.0, np.exp(-(freqs - 600) / 150))
    
    # Generiruem sluchaynye fazy i sobiraem spektr
    phases_flow = np.random.uniform(0, 2 * np.pi, len(freqs))
    flow_spectrum = pink_mag * lp_filter * np.exp(1j * phases_flow)
    
    # Perevodim obratno vo vremya
    flow = np.fft.irfft(flow_spectrum, n=N)
    flow /= np.max(np.abs(flow))  # Normalizatsiya

    # ==========================================
    # SLOY 3: THE CAVITATION (Puzyrki / Vskipanie)
    # ==========================================
    # Band-pass filtr: ostavlyaem tolko vysokie chastoty (8-12 kGts)
    bp_filter = np.where((freqs >= 8000) & (freqs <= 12000), 1.0, 0.0)
    phases_cav = np.random.uniform(0, 2 * np.pi, len(freqs))
    cav_spectrum = bp_filter * np.exp(1j * phases_cav)
    
    cav_base = np.fft.irfft(cav_spectrum, n=N)
    
    # Vozvedenie v 3-yu stepen delaet iz sploshnogo "shipeniya" ostrye impulsy (tresk lopayuschihsya puzyrkov)
    cav_crackle = cav_base ** 3
    cav_crackle /= np.max(np.abs(cav_crackle))

    # Dinamika kavitatsii: privyazyvaem k vrascheniyu lopastey vinta (naprimer, 20 oborotov v sekundu)
    prop_rpm = 20 # 20 * 5 sek = 100 tsiklov (besshovno)
    prop_lfo = (np.sin(2 * np.pi * prop_rpm * t) + 1) / 2  # Ot 0 do 1

    # Vliyanie peremennoy LOAD:
    # Pri load=1.0 (polnyy gaz) puzyrki kipyat pochti postoyanno (prop_lfo ** 0.5 - shirokie piki)
    # Pri load=0.2 (holostoy hod) puzyrki lish slegka pohrustyvayut na pikah oborotov (prop_lfo ** 4 - uzkie piki)
    envelope_sharpness = 5.0 * (1.0 - load) + 0.5 
    dynamic_envelope = prop_lfo ** envelope_sharpness

    cavitation = cav_crackle * dynamic_envelope * (load * 1.5)

    # ==========================================
    # MIKSING I MASTERING
    # ==========================================
    print("Sinhronizatsiya magnitnyh potokov...")
    
    # Svodim vse v edinyy massiv (Flow daet "telo", Hum - "motor", Cavitation - "agressiyu")
    mix = 0.9 * hum + 1.2 * flow + 2.0 * cavitation
    
    # Saturatsiya/Iskazheniya (NASA-punk stil). Priblizhaem k pryamougolnoy volne dlya plotnosti
    driven = np.tanh(mix * 2.0) / 2.0
    
    # Normalizatsiya v 16-bitnyy PCM dlya WAV
    final_signal = driven / np.max(np.abs(driven))
    audio_data = np.int16(final_signal * 32767)

    # Sohranyaem v fayl
    filename = "manta_idle_loop.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Audio-lup uspeshno sgenerirovan: {filename} (Besshovnost garantirovana)")

if __name__ == "__main__":
    # Poprobuy pomenyat load ot 0.1 do 1.0, chtoby uslyshat raznitsu v povedenii turbiny
    generate_thruster_loop(duration=5.0, load=0.85)
Glavnye fishki sinteza:

The FFT Loop Trick: V sloyah Flow i Cavitation vmesto vstroennyh generatorov numpy.random v pryamom vremeni, my berem spektr, vyrezaem iz nego nuzhnye chastoty i konvertiruem vo vremya (np.fft.irfft). Eto na 100% garantiruet, chto nachalo fayla soydetsya s ego kontsom s tochnostyu do sempla.

Eksponentsialnaya ogibayuschaya load: Vyrazhenie prop_lfo ** envelope_sharpness. Eta matematicheskaya ulovka menyaet ne prosto gromkost puzyrkov, a ih harakter. Pri vysokoy nagruzke voda vokrug vinta postoyanno vskipaet. Pri nizkoy — kavitatsiya voznikaet tolko v samyh pikovyh tochkah vrascheniya rotora (korotkimi, redkimi vspleskami).

Vozvedenie shuma v kub (cav_base ** 3): Prostoy filtrovannyy shum na 8-12 kGts zvuchit kak obychnoe shipenie zmei. No esli vozvesti volnu v tretyu stepen, tihie zvuki stanut mikroskopicheskimi, a gromkie piki sohranyatsya. V rezultate shipenie prevraschaetsya v vysokochastotnyy tresk/pesok, kotoryy idealno opisyvaet shlopyvayuschiesya pod vodoy vakuumnye puzyrki.


Etot skript prodolzhaet kontseptsiyu matematicheskoy generatsii besshovnyh industrialnyh zvukov. import numpy as np
from scipy.io import wavfile

def generate_cutter_loop(duration=5.0):
    sample_rate = 44100
    # endpoint=False garantiruet, chto posledniy sempl ne sovpadaet s pervym (vazhno dlya lupa)
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print("Aktivatsiya termalnogo rezaka. Vnimanie: zafiksirovano vneshnee davlenie vody...")

    # ==========================================
    # VSPOMOGATELNAYa FUNKTsIYa DLYa BESShOVNOGO ShUMA
    # ==========================================
    freqs = np.fft.rfftfreq(N, 1 / sample_rate)
    
    def seamless_noise(filter_mask):
        """Generiruet idealno zatsiklennyy shum zadannoy chastotnoy formy cherez IFFT"""
        phases = np.random.uniform(0, 2 * np.pi, len(freqs))
        spectrum = filter_mask * np.exp(1j * phases)
        noise = np.fft.irfft(spectrum, n=N)
        return noise / np.max(np.abs(noise))

    # ==========================================
    # SLOY 1: THE THERMAL SIZZLE (Kipenie/Isparenie)
    # ==========================================
    # High-pass filtr: ostavlyaem tolko "shipyaschie" chastoty ot 3 kGts i vyshe
    sizzle_mask = np.where(freqs > 3000, 1.0, 0.0) * np.exp(-(np.maximum(0, freqs-3000))/5000)
    raw_sizzle = seamless_noise(sizzle_mask)
    
    # Haotichnaya amplitudnaya modulyatsiya (AM) dlya imitatsii neravnomernogo kipeniya vody
    # Generiruem nizkochastotnyy "upravlyayuschiy" shum (do 30 Gts)
    am_mask = np.where(freqs < 30, 1.0, 0.0)
    am_mod = seamless_noise(am_mask)
    
    # Smeschaem modulyator v diapazon [0.2, 1.0] i vozvodim v kvadrat dlya agressivnosti pulsatsiy
    am_env = (0.6 + 0.4 * am_mod) ** 2
    thermal_sizzle = raw_sizzle * am_env

    # ==========================================
    # SLOY 2: THE STRUCTURAL RESONANCE (Vibratsiya korpusa)
    # ==========================================
    # Gudyaschie chastoty. 120, 144 i 196 Gts idealno kratny 5 sekundam (tseloe chislo tsiklov).
    base_res = (
        1.0 * np.sin(2 * np.pi * 120 * t) +
        0.6 * np.sin(2 * np.pi * 144 * t) +
        0.4 * np.sin(2 * np.pi * 196 * t)
    )
    
    # HARD CLIPPING (Zhestkoe ogranichenie). 
    # Umnozhaem sinusoidy na 4 (peregruz), a zatem zhestko srezaem verhushki.
    # Eto prevraschaet myagkiy gul v pronzitelnyy metallicheskiy skrezhet/soprotivlenie.
    resonance = np.clip(base_res * 4.0, -0.6, 0.6)
    
    # Legkoe drozhanie ruki operatora (modulyatsiya gromkosti rezonansa)
    res_mod_mask = np.where(freqs < 5, 1.0, 0.0)
    res_mod = seamless_noise(res_mod_mask)
    resonance *= (0.8 + 0.2 * res_mod)

    # ==========================================
    # SLOY 3: THE IONIZED ARC (Elektricheskaya duga / Plazma)
    # ==========================================
    # Chastoty dlya FM-modulyatsii. Nesuschaya (1500 Gts) moduliruetsya nizkochastotnym treskom (80 Gts)
    fm_carrier_freq = 1500
    fm_mod_freq = 80
    
    # Sozdaem redkie, ostrye "vspyshki" plazmy, kogda rezak natykaetsya na tugoplavkiy splav.
    # Peremnozhenie sinusoid i vozvedenie v 8-yu stepen daet rezkie impulsy.
    # 0.4 Gts i 1.8 Gts dayut rovno 2 i 9 tsiklov za 5 sekund (besshovno).
    arc_burst_env = (np.sin(2 * np.pi * 0.4 * t) * np.sin(2 * np.pi * 1.8 * t)) ** 8
    
    # Dobavlyaem haosa v ogibayuschuyu vspyshek
    arc_rnd_mask = np.where(freqs < 10, 1.0, 0.0)
    arc_rnd = seamless_noise(arc_rnd_mask)
    combined_arc_env = arc_burst_env * (0.8 + 0.5 * arc_rnd)
    
    # Indeks modulyatsii zavyazan na vspyshki (kogda vspyshka - zvuk razryvaetsya)
    mod_index = 18.0 * combined_arc_env
    
    fm_phase = 2 * np.pi * fm_carrier_freq * t + mod_index * np.sin(2 * np.pi * fm_mod_freq * t)
    ionized_arc = np.sin(fm_phase) * combined_arc_env

    # ==========================================
    # MIKSING I MASTERING (Asimmetrichnyy distorshn)
    # ==========================================
    print("Prozhig broni. Kalibrovka iskazheniy...")
    
    # Svodim sloi: kipenie daet fakturu, rezonans - "telo", arka - pronzitelnye aktsenty
    mix = 1.0 * thermal_sizzle + 0.7 * resonance + 1.4 * ionized_arc
    
    # Asimmetrichnyy peregruz (Lampovyy effekt)
    # Polozhitelnye poluvolny skruglyaem (soft-clipping), otritsatelnye - zhestko srezaem.
    # Eto sozdaet ochen "rvanyy", industrialnyy harakter zvuka.
    mix_driven = mix * 1.8
    mix_driven = np.where(mix_driven > 0, np.tanh(mix_driven), np.clip(mix_driven, -0.8, 0.8))
    
    # Finalnaya normalizatsiya
    final_signal = mix_driven / np.max(np.abs(mix_driven))
    audio_data = np.int16(final_signal * 32767)

    # Eksport
    filename = "plasma_cut_loop.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Zapis termalnogo povrezhdeniya sohranena: {filename} (5 sek, Besshovno)")

if __name__ == "__main__":
    generate_cutter_loop()

Polozhitelnye piki myagko splyuschivayutsya tangensom, a otritsatelnye bezzhalostno obrubayutsya. import numpy as np
from scipy.io import wavfile

def generate_madness_whisper(duration=7.0):
    sample_rate = 44100
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print("KRITIChESKAYa OShIBKA: Obnaruzheno nesanktsionirovannoe obraschenie k golosovomu modulyu Hecton-OS...")
    print("Rasshifrovka bityh sektorov...")

    # ==========================================
    # SLOY 1: THE UNCANNY PITCH (Lomanaya rech / Vokoder)
    # ==========================================
    # Generiruem "fonemy" — sluchaynye chastoty, prygayuschie po oktavam.
    # Oktavy (110, 220, 440, 880, 1760, 3520 Gts) dayut zloveschiy, nechelovecheskiy stroy.
    control_points = int(duration * 12)  # Skorost "razgovora" (~12 fonem v sekundu)
    octaves = np.array([110, 220, 440, 880, 1760, 3520])
    random_freqs = np.random.choice(octaves, control_points)
    
    # Interpoliruem chastoty na vsyu dlinu audio
    freq_interp = np.interp(t, np.linspace(0, duration, control_points), random_freqs)
    
    # Sglazhivaem skachki chastot skolzyaschim oknom, chtoby poluchit bystrye Chirp-skolzheniya (glissando), 
    # a ne mgnovennye pereklyucheniya. Eto zvuchit kak sintezator, pytayuschiysya "vygovorit" bukvy.
    window = np.ones(1500) / 1500
    smooth_freq = np.convolve(freq_interp, window, mode='same')
    
    # Integriruem chastotu (cumsum) dlya polucheniya nepreryvnoy fazy bez schelchkov
    phase = 2 * np.pi * np.cumsum(smooth_freq) / sample_rate
    voice_sine = np.sin(phase)
    
    # Dobavlyaem "Dyhanie/Shepot" — umnozhaem belyy shum na nashu sinusoidu.
    # Eto perenosit spektr shuma na tekuschuyu chastotu golosa (effekt primitivnogo vokodera).
    whisper_noise = np.random.randn(N)
    voice_core = 0.5 * voice_sine + 1.2 * (voice_sine * whisper_noise)

    # ==========================================
    # SLOY 2: THE GRANULAR STUTTER (Zaikanie i mikro-razryvy)
    # ==========================================
    stutter_mask = np.zeros(N)
    i = 0
    
    # Algoritm granulyarnoy narezki. My sluchaynym obrazom vklyuchaem i vyklyuchaem zvuk
    while i < N:
        # Dlina "zerna" (grain) zvuka: 10 - 30 ms
        grain_samples = int(sample_rate * np.random.uniform(0.01, 0.03))
        # Dlina mikro-pauzy: 5 - 30 ms
        pause_samples = int(sample_rate * np.random.uniform(0.005, 0.03))
        
        # 15% shans na dlinnuyu pauzu (imitatsiya razdeleniya na "slova")
        if np.random.rand() > 0.85:
            pause_samples += int(sample_rate * np.random.uniform(0.1, 0.4))
            
        end_idx = min(i + grain_samples, N)
        grain_length = end_idx - i
        
        if grain_length > 0:
            # Sluchaynaya gromkost dlya kazhdoy granuly (effekt rvanoy lenty)
            env = np.ones(grain_length) * np.random.uniform(0.2, 1.0)
            
            # Dobavlyaem kroshechnyy fade-in/fade-out (1 ms) po krayam granuly,
            # chtoby izbezhat DC-schelchkov, ostavlyaya tolko "tsifrovoy" ritm.
            f_len = min(int(sample_rate * 0.001), grain_length // 2)
            if f_len > 0:
                fade = np.linspace(0, 1, f_len)
                env[:f_len] *= fade
                env[-f_len:] *= fade[::-1]
                
            stutter_mask[i:end_idx] = env
            
        i += grain_samples + pause_samples
        
    stuttered_voice = voice_core * stutter_mask

    # ==========================================
    # SLOY 3: THE RING MODULATION (Metallicheskiy tembr)
    # ==========================================
    # Generiruem sverh-nizkochastotnyy shum (okolo 25 Gts) s pomoschyu interpolyatsii
    lf_points = int(duration * 25)
    lf_noise = np.interp(t, np.linspace(0, duration, lf_points), np.random.uniform(-1, 1, lf_points))
    
    # Smeshivaem nizkochastotnyy shum s chistym sinusom na 55 Gts (daet klassicheskiy SCI-FI metall)
    rm_modulator = lf_noise + 0.6 * np.sin(2 * np.pi * 55 * t)
    
    # Ring Modulation: peremnozhaem granulyarnyy golos na modulyator
    alien_whisper = stuttered_voice * rm_modulator

    # ==========================================
    # MIKSING I MASTERING (Unichtozhenie signala)
    # ==========================================
    print("Formirovanie audio-gallyutsinatsii...")
    
    # Ekstremalnyy Hard-clipping. Razgonyaem signal v 6 raz i zhestko obrubaem.
    # Eto prevraschaet shepot v agressivnyy tsifrovoy "skrezhet".
    driven_signal = np.clip(alien_whisper * 6.0, -0.9, 0.9)
    
    # Finalnaya normalizatsiya
    final_signal = driven_signal / np.max(np.abs(driven_signal))
    audio_data = np.int16(final_signal * 32767)

    # Eksport
    filename = "ai_whisper_glitch.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Damp pamyati sohranen. Proslushayte {filename} (Na svoy strah i risk)")

Razbor mehaniki zvuka (Psihoakustika bezumiya):

Fazovoe integrirovanie (Sloy 1): Samaya chastaya oshibka pri sinteze menyayuschihsya chastot (Chirp) — eto prosto peredat massiv chastot v np.sin(2 * np.pi * freq * t). Eto vyzovet zhutkie schelchki pri kazhdom perepade, potomu chto faza sbrasyvaetsya. My ispolzuem metod np.cumsum(smooth_freq), kotoryy matematicheski nakaplivaet ugol. Eto zastavlyaet II plavno «tyanut» (skolzit) glasnye zvuki, dazhe kogda on prygaet na 3 oktavy vverh.

Amplitudnyy Vokoder: II ne mozhet «dyshat». Chtoby sdelat golos pohozhim na shepot, my umnozhaem sinusoidu na belyy shum. Eto kontsentriruet spektr shuma rovno na chastote sinusoidy. V rezultate my slyshim ne pisk, a skrezheschuschee shipenie, menyayuschee svoyu tonalnost.

While-loop dlya Granulyarnogo Sinteza: My ne ispolzuem standartnye LFO (sinusoidy) dlya effekta zaikaniya, potomu chto oni predskazuemy. Tsikl while narezaet zvuk asimmetrichnymi kuskami. Vash mozg bezuspeshno pytaetsya nayti ritm v etih 10-30 ms obryvkah, chto vyzyvaet chuvstvo trevogi.

Ring Modulation (Koltsevaya modulyatsiya): Pri umnozhenii zvuka na 55 Gts + NCh-shum, poyavlyayutsya tak nazyvaemye bokovye polosy (sidebands) chastot, kotorye matematicheski ne kratny osnovnomu tonu. Imenno eta nekratnost ubivaet chelovechnost v golose, pridavaya emu znamenityy metallicheskiy okras (vspomni golosa Dalekov iz Doktora Kto, oni byli sdelany po tochno takoy zhe tehnologii, tolko analogovoy).

def generate_exertion_breath(duration=5.0):
    sample_rate = 44100
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print("KRITIChESKIY SBOY: Davlenie v konture zhizneobespecheniya padaet...")
    print("Generatsiya akusticheskogo profilya gipoksii...")

    # ==========================================
    # SLOY 1: THE INHALE/EXHALE (Formanty i potok vozduha)
    # ==========================================
    # Uskoryaem dyhanie k kontsu (ot 0.35 Gts do 0.7 Gts)
    breath_rate = np.linspace(0.35, 0.7, N)
    breath_phase = 2 * np.pi * np.cumsum(breath_rate) / sample_rate
    
    # Sozdaem ogibayuschie vdoha i vydoha. Smeschenie -0.15 sozdaet mertvuyu zonu (pauzu v dyhanii)
    inhale_env = np.maximum(0, np.sin(breath_phase) - 0.15)
    exhale_env = np.maximum(0, -np.sin(breath_phase) - 0.15)
    
    # Iskazhaem krivuyu: vdoh rezkiy i obryvistyy (panika), vydoh tyazhelyy
    inhale_env = (inhale_env / np.max(inhale_env)) ** 1.5
    exhale_env = (exhale_env / np.max(exhale_env)) ** 2.0

    # Generatsiya formant (rezonansov) cherez chastotnye maski (FFT)
    freqs = np.fft.rfftfreq(N, 1 / sample_rate)
    
    # Maska vdoha: Uzkoe gorlo, spazm. Rezonans na 1200 Gts i shipenie zubov na 2800 Gts
    inhale_mask = 1.5 * np.exp(-((freqs - 1200) / 300)**2) + 0.8 * np.exp(-((freqs - 2800) / 600)**2)
    
    # Maska vydoha: Vnutri shlema. Gluhoy "bochkoobraznyy" rezonans 500 Gts i 1400 Gts
    exhale_mask = 2.0 * np.exp(-((freqs - 500) / 200)**2) + 0.6 * np.exp(-((freqs - 1400) / 400)**2)
    
    def generate_formant_noise(mask):
        phases = np.random.uniform(0, 2 * np.pi, len(freqs))
        return np.fft.irfft(mask * np.exp(1j * phases), n=N)

    inhale_noise = generate_formant_noise(inhale_mask)
    exhale_noise = generate_formant_noise(exhale_mask)
    
    # Dobavlyaem "drozh/hrip" na vdohe — amplitudnaya modulyatsiya na 12 Gts (spazm diafragmy)
    tremolo = 1.0 + 0.4 * np.sin(2 * np.pi * 12 * t) * inhale_env
    
    airflow = (inhale_noise * inhale_env + exhale_noise * exhale_env) * tremolo
    airflow /= np.max(np.abs(airflow))

    # ==========================================
    # SLOY 2: THE VALVE CLICK (Mehanika klapanov maski)
    # ==========================================
    valves = np.zeros(N)

    def get_triggers(env, threshold=0.05):
        is_active = (env > threshold).astype(int)
        crossings = np.where(np.diff(is_active) == 1)[0]
        return crossings

    all_starts = np.sort(np.concatenate([get_triggers(inhale_env), get_triggers(exhale_env)]))

    for idx in all_starts:

        if idx + click_len < N:
            t_click = np.linspace(0, 0.02, click_len, endpoint=False)
            # Rezkiy mehanicheskiy schelchok: spad gromkosti + rezonans pruzhiny klapana (1200 Gts) + shum
            env_click = np.exp(-t_click * 400)
            click_sig = env_click * (np.sin(2 * np.pi * 1200 * t_click) + 0.6 * np.random.randn(click_len))
            valves[idx:idx+click_len] += click_sig
            
    valves /= (np.max(np.abs(valves)) + 1e-9)

    # ==========================================
    # SLOY 3: THE HEARTBEAT (Sub-bas)
    # ==========================================
    # Puls uskoryaetsya s 90 do 168 udarov v minutu (ot 1.5 do 2.8 Gts)
    heart_bpm_hz = np.linspace(1.5, 2.8, N)
    heart_phase = 2 * np.pi * np.cumsum(heart_bpm_hz) / sample_rate
    
    # Formiruem dvoynoy udar (lub-dub). 
    # Vozvedenie v 30-yu stepen suzhaet shirokuyu sinusoidu do ostryh korotkih pikov.
    lub = np.maximum(0, np.sin(heart_phase)) ** 30
    dub = np.maximum(0, np.sin(heart_phase - 1.2)) ** 30 * 0.6 # Smeschenie fazy daet vtoroy udar
    heart_env = lub + dub
    
    # Nesuschaya chastota serdtsa (40 Gts) daet oschutimyy tolchok v sabvufer.
    heart_carrier = np.sin(2 * np.pi * 40 * t)
    
    # Sglazhennyy NCh-shum dobavlyaet gluhoy "udar v grudnuyu kletku"
    heart_noise = np.convolve(np.random.randn(N), np.ones(100)/100, mode='same') 
    
    heartbeat = heart_env * (heart_carrier + 0.4 * heart_noise)
    heartbeat /= np.max(np.abs(heartbeat))

    # ==========================================
    # MIKSING I MASTERING (Vnutrennosti skafandra)
    # ==========================================
    print("Kompilyatsiya biometricheskih dannyh...")
    
    # Klapany i dyhanie zvuchat gromko, a serdtse pulsiruet "pod" nimi
    mix = 1.0 * airflow + 0.4 * valves + 1.2 * heartbeat

    # Splyuschivanie sinusoidy (tanh) imitiruet peregruz deshevogo mikrofona vnutri shlema,
    # kotoryy ne spravlyaetsya s agressivnym davleniem vozduha.
    driven = np.tanh(mix * 1.5) / 1.5
    
    final_signal = driven / np.max(np.abs(driven))
    audio_data = np.int16(final_signal * 32767)

    print(f"Gotovo. Uroven paniki sohranen: {filename}")

Razbor mehaniki zvuka (Akusticheskaya psihologiya):

Integratsiya chastoty (np.cumsum): Chtoby serdtse i dyhanie uskoryalis besshovno, my ne mozhem prosto ispolzovat formulu sin(f * t), tak kak pri izmenenii f faza razorvetsya. Vmesto etogo my integriruem chastotu kadr za kadrom (nakoplenie fazy). Serdtse plavno sryvaetsya s 90 udarov v minutu do pochti kriticheskih 168.

Formantnye krivye (np.exp v chastotnoy oblasti): Vmesto slozhnyh kaskadov IIR-filtrov, my stroim v spektre Gaussovy «kolokola» na chastotah 1200 i 2800 Gts. Propuskaya cherez nih belyy shum s pomoschyu obratnogo preobrazovaniya Fure (irfft), my fizicheski modeliruem prohozhdenie vozduha cherez uzkuyu schel gortani, ne ispolzuya semply realnogo golosa.

Hrip diafragmy (tremolo): Pri silnoy panike myshtsy svodit sudorogoy. Amplitudnaya modulyatsiya v 12 Gts, nalozhennaya tolko na fazu inhale_env, zastavlyaet vozduh preryvatsya («zapinatsya»), chto mgnovenno schityvaetsya nashim mozgom kak kraynyaya stepen stressa.

Algoritm triggerov (get_triggers): Chtoby klapany schelkali strogo v tot moment, kogda podaetsya vozduh, skript analiziruet ogibayuschuyu dyhaniya, nahodit peresecheniya nulevoy otmetki s pomoschyu funktsii raznitsy massiva np.diff i matematicheski "strelyaet" 20-millisekundnym klikom pruzhiny reduktora.

import numpy as np
from scipy.io import wavfile

def generate_mech_impact(duration=3.0):
    sample_rate = 44100
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print("Registratsiya seysmicheskogo tolchka. Raschet massy obekta...")

    # ==========================================
    # SLOY 1: THE LOW THUD (Massa i Udar)
    # ==========================================
    # Imitatsiya massy cherez bystroe padenie chastoty (Pitch Envelope).
    # Chastota startuet s 200 Gts i za dolyu sekundy padaet do sab-basa v 40 Gts.
    freq_drop = 40 + 160 * np.exp(-t * 40)
    
    # Integriruem chastotu dlya polucheniya nepreryvnoy fazy
    phase = 2 * np.pi * np.cumsum(freq_drop) / sample_rate
    
    # Ogibayuschaya udara: mgnovennaya ataka i plotnoe, no korotkoe zatuhanie
    thud_env = np.exp(-t * 6)
    
    # Dobavlyaem zhestkuyu saturatsiyu uzhe na etape generatsii, chtoby "rasplyuschit" sinus
    # i poluchit plotnyy pryamougolno-podobnyy panch.
    raw_thud = np.sin(phase) * thud_env
    heavy_thud = np.tanh(raw_thud * 5.0) / 2.0

    # ==========================================
    # SLOY 2: THE MECHANICAL RING (Lyazg i zvon konstruktsiy)
    # ==========================================
    # Zvon metalla ne byvaet garmonichnym. My berem nekratnye chastoty v diapazone 400-800 Gts.
    metal_carrier = (
        1.0 * np.sin(2 * np.pi * 412 * t) +
        0.7 * np.sin(2 * np.pi * 538 * t) +
        0.4 * np.sin(2 * np.pi * 781 * t)
    )
    
    # Ring Modulation: umnozhaem na nizkuyu chastotu (140 Gts). 
    # Eto sozdaet intermodulyatsionnye iskazheniya — znamenityy "vedernyy" ili "robotizirovannyy" tembr.
    ring_modulator = np.sin(2 * np.pi * 140 * t)
    
    # Rezkaya ataka dlya lyazga. Zvon metalla zatuhaet bystree, chem nizkochastotnyy gul.
    ring_env = np.exp(-t * 12)
    mechanical_ring = metal_carrier * ring_modulator * ring_env

    # ==========================================
    # SLOY 3: THE SILT DISPLACEMENT (Vsplesk ila i peska)
    # ==========================================
    # Generiruem Rozovyy shum (spad 3 dB na oktavu) cherez FFT, 
    # on zvuchit bolee massivno i "temno" po sravneniyu s belym.
    freqs = np.fft.rfftfreq(N, 1 / sample_rate)
    
    # Zaschita ot deleniya na nol dlya pervoy chastoty (DC)
    pink_mag = np.zeros_like(freqs)
    pink_mag[1:] = 1.0 / np.sqrt(freqs[1:])
    
    # Low-pass filtr: il tyazhelyy, poetomu srezaem vse vyshe 400 Gts, ostavlyaya tolko nizkiy gul potoka
    lp_filter = np.where(freqs <= 400, 1.0, np.exp(-(freqs - 400) / 200))
    
    phases = np.random.uniform(0, 2 * np.pi, len(freqs))
    spectrum = pink_mag * lp_filter * np.exp(1j * phases)
    silt_noise = np.fft.irfft(spectrum, n=N)
    
    # Normalizatsiya shuma pered nalozheniem ogibayuschey
    silt_noise /= np.max(np.abs(silt_noise))
    
    # Ogibayuschaya ila: narastaet za 50ms (poka pesok podnimaetsya so dna) 
    # i ochen medlenno osedaet.
    silt_env = np.clip(t / 0.05, 0, 1) * np.exp(-t * 2.5)
    silt_displacement = silt_noise * silt_env

    # ==========================================
    # MIKSING I MASTERING (Akustika tolschi vody)
    # ==========================================
    print("Kompilyatsiya zvukovyh sloev i fiziki sredy...")
    
    # Smeshivaem. Thud daet udar v grud, Ring daet metall, Silt daet realizm okruzheniya.
    mix = 1.5 * heavy_thud + 0.9 * mechanical_ring + 1.2 * silt_displacement
    
    # Ekstremalnaya kompressiya (Soft-Clipping).
    # Umnozhaem na 3.0, chtoby "vpechatat" zvuk v potolok, a tanh sgladit ugly, 
    # sozdavaya oschuschenie chudovischnogo perepada davleniya v vode pri udare.
    driven_impact = np.tanh(mix * 3.0) / 1.5
    
    # Finalnaya normalizatsiya
    final_signal = driven_impact / np.max(np.abs(driven_impact))
    audio_data = np.int16(final_signal * 32767)

    # Eksport
    filename = "heavy_mech_thud.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Analiz zavershen. Akusticheskiy sled zapisan: {filename}")

if __name__ == "__main__":
    generate_mech_impact()
Razbor mehaniki zvuka (Fizika udara):

Effekt Inertsii (freq_drop): Sinus, kotoryy prosto igraet na 40 Gts, zvuchit kak gul transformatora. No esli zastavit chastotu eksponentsialno upast s 200 Gts do 40 Gts za pervye millisekundy, nashe uho vosprinimaet eto kak "udar", fizicheskiy tolchok massivnogo obekta. Etot priem (Pitch Envelope) ispolzuetsya dlya sinteza vseh barabanov (Kicks) v elektronnoy muzyke.

Dinamicheskiy peregruz (np.tanh(raw_thud * 5.0)): My splyuschivaem chistyy sinus osnovnogo udara esche do svedeniya so vsem ostalnym miksom. Eto delaet «telo» udara tolstym, kvadratnym (po forme volny) i neveroyatno plotnym, imitiruya gidrodinamicheskiy udar v vode.

Nekratnyy metall: Polye metallicheskie predmety (kak bronya meha) nikogda ne rezoniruyut chistymi akkordami. My ispolzuem chastoty 412, 538, 781 Gts — oni matematicheski ne sootnosyatsya drug s drugom (dissonans). A peremnozhenie ih na 140 Gts (Ring Mod) drobit eti chastoty esche silnee, generiruya gryaznyy industrialnyy zvon.

Smeschenie ila (Ataka ogibayuschey): Funktsiya np.clip(t / 0.05, 0, 1) sozdaet lineynoe narastanie gromkosti rovno za 50 millisekund (fade-in), posle chego vstupaet v silu spadayuschaya eksponenta. V otlichie ot metalla i udara (kotorye proishodyat mgnovenno v nulevuyu sekundu), il imeet mikro-zaderzhku v raskrytii zvuka. Eta kroshechnaya raznitsa vo vremeni ataki daet mozgu ponyat, chto zvuk imeet masshtab i fiziku: Snachala udar -> potom podnimaetsya oblako.

import numpy as np
from scipy.io import wavfile

def generate_emp_surge(duration=4.0):
    sample_rate = 44100
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)

    print("VNIMANIE: Zafiksirovan proboy ekranirovaniya. Vybros EMI...")

    # ==========================================
    # SLOY 1: THE STATIC BURST (Razryad i tsifrovoy tresk)
    # ==========================================
    # Dlya sozdaniya effekta "umirayuschego protsessora" my ispolzuem effekt Sample-and-Hold (S&H).
    # Chastota diskretizatsii (kvantovaniya) shuma nachinaetsya s 15000 Gts i stremitelno padaet do 50 Gts.
    sh_freq = 50 + 15000 * np.exp(-t * 5)
    
    # Integriruem chastotu, chtoby poluchit fazu semplirovaniya
    sh_phase = np.cumsum(sh_freq) / sample_rate
    
    # Prevraschaem fazu v tselye indeksy. Eto vektorizovannyy matematicheskiy tryuk:
    # massiv indices budet vyglyadet kak[0,0,1,1,1,2,2, ...], sozdavaya "stupenki" vo vremeni.
    indices = np.floor(sh_phase).astype(int)
    
    # Generiruem sluchaynyy belyy shum i "rastyagivaem" ego po indeksam S&H
    random_values = np.random.randn(indices[-1] + 2)
    sh_noise = random_values[indices]
    
    # Ogibayuschaya: moschneyshaya ataka v nachale, zatuhayuschaya za 1.5 sekundy
    burst_env = np.exp(-t * 4)
    static_burst = sh_noise * burst_env

    # ==========================================
    # SLOY 2: THE POWER DOWN (Zatuhanie turbin / kondensatorov)
    # ==========================================
    # Glissando: stremitelnoe padenie chastoty ot 1000 Gts do 50 Gts.
    freq_drop = 50 + 950 * np.exp(-t * 3.5)
    
    # Nakoplenie fazy (cumsum) dlya plavnogo spada bez schelchkov
    phase_drop = 2 * np.pi * np.cumsum(freq_drop) / sample_rate
    
    # Sozdaem sinusoidu, no slegka peregruzhaem ee, chtoby zvuk byl bolee sinteticheskim
    power_down_raw = np.sin(phase_drop)
    power_down_env = np.exp(-t * 2)
    power_down = power_down_raw * power_down_env

    # ==========================================
    # SLOY 3: THE HIGH-VOLTAGE HUM (Otklyuchenie pitaniya)
    # ==========================================
    # Ispolzuem funktsiyu np.sign, chtoby prevratit sinus v KVADRATNUYu volnu (Square Wave).
    # Kvadratnaya volna soderzhit beskonechnoe kolichestvo nechetnyh garmonik, sozdavaya "zhuzhzhanie" toka.
    # Smeshivaem 60 Gts i 50 Gts dlya dissonansa (neispravnyy transformator).
    hum_60 = np.sign(np.sin(2 * np.pi * 60 * t))
    hum_50 = np.sign(np.sin(2 * np.pi * 50 * t))
    voltage_hum_base = 0.6 * hum_60 + 0.4 * hum_50
    
    # Sozdaem "agoniyu" transformatora pered otklyucheniem.
    # Do 0.4 sekundy gul rovnyy, zatem nachinaet zapinatsya (moduliruetsya 15 Gts),
    # a posle 0.8 sek - mgnovenno umiraet (zhestkaya eksponenta).
    stutter = np.maximum(0, np.sign(np.sin(2 * np.pi * 15 * t))) # Vkl/Vykl 15 raz v sekundu
    
    hum_env = np.ones(N)
    cutoff_point = int(sample_rate * 0.8)
    hum_env[:cutoff_point] = 0.7 + 0.3 * stutter[:cutoff_point]
    hum_env[cutoff_point:] = np.exp(-(np.linspace(0, duration - 0.8, N - cutoff_point)) * 40)
    
    voltage_hum = voltage_hum_base * hum_env

    # ==========================================
    # SLOY 4: THE SUB-PUNCH (Sam udar volny EMI)
    # ==========================================
    # Kroshechnyy, no sverhmoschnyy sub-basovyy hlopok v samom nachale (0-50ms)
    pop_phase = 2 * np.pi * np.cumsum(80 * np.exp(-t * 60)) / sample_rate
    emp_pop = np.sin(pop_phase) * np.exp(-t * 40) * 2.0

    # ==========================================
    # MIKSING I MASTERING (Unichtozhenie dinamiki)
    # ==========================================
    print("Vyzhiganie mikroshem...")
    
    # Sobiraem vse sloi
    mix = 1.2 * static_burst + 1.0 * power_down + 0.5 * voltage_hum + emp_pop
    
    # Zhestkiy limiter i saturatsiya.
    # Umnozhenie na 4 pered tanh zagonyaet signal v ekstremalnyy peregruz,
    # chto imitiruet fizicheskoe peregoranie audio-trakta ot vysokogo napryazheniya.
    driven_mix = np.tanh(mix * 4.0) / 2.0
    
    # Finalnaya normalizatsiya
    final_signal = driven_mix / np.max(np.abs(driven_mix))
    audio_data = np.int16(final_signal * 32767)

    filename = "emp_blast.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Tsepi obestocheny. Audio-otchet sohranen: {filename}")

if __name__ == "__main__":
    generate_emp_surge()
Razbor mehaniki zvuka (Akustika smerti elektroniki):

Vektorizovannyy Sample-and-Hold (indices = np.floor(sh_phase)): Eto neveroyatno moschnyy matematicheskiy priem. Vmesto togo chtoby filtrovat belyy shum ekvalayzerami, my obmanyvaem vremya. V nachale skripta shum sempliruetsya na chastote 15 kGts (zvuchit kak obychnoe shipenie). No k kontsu sekundy chastota semplirovaniya padaet do 50 Gts. Zvuk razbivaetsya na ogromnye tsifrovye "pikseli" (stupenki), prevraschayas v zhestkiy robotizirovannyy skrezhet (Bitcrush/Decimation). Eto zvuk togo, kak protsessoru ne hvataet taktovoy chastoty.

Kvadratnye volny transformatora (np.sign): Obychnaya elektroset gudit sinusoidalno. No kogda tsep probivaet, napryazhenie prygaet ryvkami. Funktsiya np.sign beret plavnuyu sinusoidu i prevraschaet ee v zhestkie pryamougolniki iz 1 i -1. Matematicheski, kvadratnaya volna neset v sebe gigantskoe kolichestvo nechetnyh garmonik (slyshimoe "zuzzhanie"), kotoroe idealno lozhitsya v industrialnyy NASA-punk.

Golosovoe zaikanie seti (stutter): Gul otrubaetsya ne srazu. Mezhdu 0.0 i 0.8 sekundy my umnozhaem gul transformatora na druguyu kvadratnuyu volnu (15 Gts), znacheniya kotoroy koleblyutsya mezhdu 0 i 1 (np.maximum). Eto sozdaet korotkie mikro-razryvy pitaniya — slovno avtomatika otchayanno schelkaet rele, pytayas perezapustit sistemu, prezhde chem okonchatelno umeret.

Sverhbystryy Pitch Drop (emp_pop): EMI — eto volna nevidimoy energii, no ona bet fizicheski. Sekretnyy 4-y sloy sozdaet hlopok (Punch), chastota kotorogo padaet

Chtoby ne ispolzovat medlennye for-tsikly dlya imitatsii rezonansnogo filtra, ya primenil izyaschnyy DSP-tryuk: Sintez uzkopolosnogo shuma cherez amplitudnuyu modulyatsiyu. My generiruem sverhnizkiy gul (do 5 Gts) i matematicheski "sdvigaem" ego na plavayuschuyu chastotu (150-300 Gts) s pomoschyu peremnozheniya na sinusoidu. Eto daet 100% tochnyy effekt zavyvaniya vetra v trubah ili gulyayuschego rezonansa podvodnyh pescher.

import numpy as np
from scipy.io import wavfile

def generate_abyss_drone(duration=10.0):
    sample_rate = 44100
    t = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
    N = len(t)
    freqs = np.fft.rfftfreq(N, 1 / sample_rate)

    print("Pogruzhenie na 4000 metrov. Kavernoznye rezonansy aktivny...")

    # ==========================================
    # VSPOMOGATELNAYa FUNKTsIYa: FFT-Generator LFO (Perlin-like)
    # ==========================================
    def make_smooth_lfo(cutoff_hz, min_val, max_val):
        """Generiruet idealno gladkiy, medlennyy sluchaynyy signal (pohozhiy na Perlin Noise)"""
        noise_spec = np.fft.rfft(np.random.randn(N))
        # Zhestkiy srez vsego, chto vyshe cutoff_hz (ostavlyaem tolko medlennye kolebaniya)
        mask = np.where(freqs < cutoff_hz, 1.0, 0.0)
        lfo = np.fft.irfft(noise_spec * mask, n=N)
        # Normalizatsiya v diapazon [min_val, max_val]
        lfo = (lfo - np.min(lfo)) / (np.max(lfo) - np.min(lfo))
        return min_val + lfo * (max_val - min_val)

    # ==========================================
    # SLOY 1: THE DEEP LOOM (Infrazvukovoy Gul i Bieniya)
    # ==========================================
    # Bazovye chastoty: 25, 31, 37 Gts (Oni sozdayut temnyy, misticheskiy akkord)
    # Dobavlyaem k kazhdoy chastote mikro-smeschenie (detune), chtoby oni "perelivalis" (Phasing)
    f_bases = [25.0, 31.0, 37.0]
    detunes =[0.15, -0.12, 0.18]
    
    loom_raw = np.zeros(N)
    for f, df in zip(f_bases, detunes):
        # Osnovnoy infrazvuk
        loom_raw += np.sin(2 * np.pi * f * t) + np.sin(2 * np.pi * (f + df) * t)
        # Dobavlyaem vtoruyu garmoniku (oktavu vyshe), chtoby gul byl slyshen dazhe na noutbukah
        loom_raw += 0.4 * np.sin(2 * np.pi * (f * 2) * t)
        
    # Legkiy peregruz (saturatsiya) dobavlyaet gulu fizicheskoy plotnosti
    deep_loom = np.tanh(loom_raw * 1.5) / 1.5

    # ==========================================
    # SLOY 2: THE DISTANT MOAN (Ston Bezdny)
    # ==========================================
    # 1. Generiruem plavayuschuyu chastotu filtra (ot 150 do 300 Gts), menyaetsya kazhdye 2-3 sekundy
    wandering_freq = make_smooth_lfo(0.4, 150, 300)
    phase_moan = 2 * np.pi * np.cumsum(wandering_freq) / sample_rate
    
    # 2. Generiruem "vozduh" dlya filtra (ochen nizkochastotnyy shum s plavnym spadom)
    noise_spec = np.fft.rfft(np.random.randn(N))
    bandpass_width = np.exp(-freqs / 8.0) # Myagkiy spad k 15-20 Gts
    breathy_noise = np.fft.irfft(noise_spec * bandpass_width, n=N)
    breathy_noise /= np.max(np.abs(breathy_noise))
    
    # 3. MAGIYa DSP: Peremnozhaem plavayuschuyu sinusoidu na NCh-shum. 
    # Eto perenosit spektr shuma na chastotu sinusoidy, sozdavaya idealnuyu illyuziyu 
    # bluzhdayuschego rezonansnogo filtra s vysokim Q (effekt voyuschey truby)
    moan_resonant = breathy_noise * np.sin(phase_moan)
    
    # 4. Fonovyy Korichnevyy shum (Brownian Noise) — massivnoe dvizhenie vodnoy tolschi
    brown_mask = 1.0 / (np.maximum(freqs, 1.0) ** 1.5)
    brown_noise = np.fft.irfft(noise_spec * brown_mask, n=N)
    brown_noise /= np.max(np.abs(brown_noise))

    distant_moan = 2.0 * moan_resonant + 0.8 * brown_noise

    # ==========================================
    # SLOY 3: THE WATER WEIGHT (Davlenie Sredy)
    # ==========================================
    # Ochen medlennaya amplitudnaya modulyatsiya (tsikl ~4-5 sekund),
    # kotoraya zastavlyaet ves zvuk "nakatyvat" i "otstupat", imitiruya techeniya
    pressure_lfo = make_smooth_lfo(0.2, 0.4, 1.0)

    # ==========================================
    # MIKSING I MASTERING (Pogruzhenie vo tmu)
    # ==========================================
    print("Sintez davleniya tolschi vody...")
    
    # Smeshivaem Gul i Ston, propuskaya ih cherez modulyator Davleniya
    mix = (1.2 * deep_loom + 1.0 * distant_moan) * pressure_lfo
    
    # Plenka (Soft-clipping). Ne daet pikam probit 0 dB i "skleivaet" infrazvuk s shumom
    driven_mix = np.tanh(mix * 2.0) / 2.0
    
    # Ubiraem DC-offset (smeschenie konstanty), kotoroe moglo vozniknut pri saturatsii
    driven_mix -= np.mean(driven_mix)
    
    # Finalnaya normalizatsiya
    final_signal = driven_mix / np.max(np.abs(driven_mix))
    audio_data = np.int16(final_signal * 32767)

    filename = "deep_void_drone.wav"
    wavfile.write(filename, sample_rate, audio_data)
    print(f"Atmosfera Bezdny sgenerirovana: {filename} (Nadente naushniki)")

if __name__ == "__main__":
    generate_abyss_drone()

Phasing (Bieniya): Infrazvuk v skripte zvuchit ne rovno. Peremnozhenie i slozhenie chastot (naprimer, 25 Gts i 25.15 Gts) zastavlyaet ih to matematicheski skladyvatsya (gromko), to vzaimounichtozhatsya (tiho). Etot tsikl zanimaet neskolko sekund. Mozg schityvaet etu pulsatsiyu kak "dyhanie" chego-to gigantskogo vo tme.

SSB Sintez vmesto filtrov: Funktsiya distant_moan. Vmesto togo, chtoby sozdavat shum i "rezat" ego medlennymi ekvalayzerami, my sozdaem chastotu (wandering_freq), na kotoroy dolzhen zvuchat ston, i obvolakivaem ee mikroskopicheskim shumom (breathy_noise). Chtoby zastavit davlenie i ston plavno i nepredskazuemo menyatsya (kak Perlin Noise), my generiruem shum v spektralnoy oblasti (rfft), otrubaem vse, chto vyshe 0.4 Gts, i konvertiruem obratno vo vremya (irfft). Poluchaetsya idealnaya, organicheskaya krivaya, po kotoroy dvizhetsya vsya atmosfernaya zhizn treka.
