private static float ResolveMinimumSubmergedDepthMeters(in ScatterRuntimeRuleEntry runtimeRule)

    float authored = Mathf.Max(0f, runtimeRule.MinDepthMeters);
    if (authored > 0f)

    switch (runtimeRule.ProceduralDomain)
    {
        case WorldPrefabFamilyProfile.ProceduralDomain.Rock:
        case WorldPrefabFamilyProfile.ProceduralDomain.RockCluster:
            return 1f;
        case WorldPrefabFamilyProfile.ProceduralDomain.SafePocket:
        case WorldPrefabFamilyProfile.ProceduralDomain.ResourcePocket:

            return 2f;
        default:
            return 1.5f;
    }
}
